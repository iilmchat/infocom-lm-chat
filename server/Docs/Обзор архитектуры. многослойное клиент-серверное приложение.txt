InfocomChat.Server/
├── 📁 Properties/                      # Свойства проекта (launchSettings.json)

│   ├── AdminController.cs              # Административные функции (управление пользователями, модерация)
│   ├── ChatController.cs               # Основной чат (отправка, редактирование, история)
│   ├── MentionController.cs            # Управление упоминаниями (@username)
│   ├── PrivateChatController.cs        # Приватные чаты (личные сообщения)
│   ├── ReactionController.cs           # Реакции на сообщения (👍❤️😄)
│   ├── RoomController.cs               # Управление комнатами (создание, вход, выход)
│   └── SyncController.cs               # Синхронизация через Long Polling
│
├── 📁 Services/                        # ⚙️ БИЗНЕС-ЛОГИКА (Сервисный слой)

│   ├── PrivateChatService.cs           # Приватные чаты (логика личных сообщений)
│   ├── UserPresenceService.cs          # Отслеживание присутствия пользователей
│   ├── ReactionService.cs              # Реакции на сообщения
│   ├── AttachmentService.cs            # Работа с вложениями (файлы)
│   └── MentionService.cs               # Обработка упоминаний
│
├── 📁 Database/                        # 🗄️ ДОСТУП К ДАННЫМ (Слой персистентности)
│   ├── ChatDbContext.cs                # Контекст Entity Framework Core
│   └── DatabaseConfig.cs               # Конфигурация БД (поддержка SQL Server, PostgreSQL, SQLite)
│
├── 📁 Models/                          # 📦 МОДЕЛИ ДАННЫХ
│   ├── 📁 Entities/                    # Сущности базы данных (Entity Framework)
│   │   ├── User.cs                     # Пользователь
│   │   ├── Room.cs                     # Комната
│   │   ├── Message.cs                  # Сообщение
│   │   ├── PrivateChat.cs              # Приватный чат

│   │   ├── ModerationLog.cs            # Логи модерации
│   │   ├── SystemSetting.cs            # Системные настройки
│   │   ├── MessageReaction.cs          # Реакции на сообщения
│   │   ├── Attachment.cs               # Вложения
│   │   └── Mention.cs                  # Упоминания
│   │
│   ├── ChatMessage.cs                  # DTO для сообщений чата
│   ├── PrivateMessage.cs               # DTO для приватных сообщений
│   ├── RoomInfo.cs                     # Информация о комнате (с настройками)
│   ├── UserInfo.cs                     # Информация о пользователе
│   └── AdminModels.cs                  # Административные модели
│
├── 📁 DTOs/                            # 📨 DTO (Data Transfer Objects)
│   ├── PaginationDto.cs                # Пагинация (запрос/ответ)
│   ├── UserDtos.cs                     # DTO для пользователей
│   ├── PrivateChatDtos.cs              # DTO для приватных чатов
│   └── ReactionDtos.cs                 # DTO для реакций
│

│   └── DtoExtensions.cs                # Вспомогательные методы

│

├── 📁 Extensions/                      # 🔩 МЕТОДЫ РАСШИРЕНИЯ

│
├── 📁 Configuration/                   # ⚙️ КОНФИГУРАЦИЯ
│   ├── AppSettings.cs                  # Типизированные настройки
│   └── CorsConfiguration.cs            # Настройки CORS
│
├── 📁 Validators/                      # ✅ ВАЛИДАТОРЫ
│   └── CustomValidators.cs             # Кастомные валидаторы
│
├── Program.cs                          # 🚀 ТОЧКА ВХОДА (Настройка хоста и логирования)

├── appsettings.json                    # 📄 КОНФИГУРАЦИЯ (БД, CORS, настройки чата)
